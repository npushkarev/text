APP=$(pgrep -f 'BitbucketServerLauncher' | head -1); echo "APP PID=$APP"

echo "=== TLS/SSL/cipher флаги ЖИВОГО процесса Bitbucket ==="
tr '\0' '\n' < /proc/$APP/cmdline \
  | grep -iE 'mail\.smtp|jdk\.tls|\bssl|cipher|namedgroup|ecdh|curve|crypto|\.protocols'

echo "=== те же флаги в стартовых скриптах ==="
grep -rniE 'JVM_SUPPORT_RECOMMENDED_ARGS|mail\.smtp|jdk\.tls|cipher|namedGroup|ssl\.protocols|namedgroups' \
  /opt/atlassian/bitbucket/5.13.1/bin/ 2>/dev/null
