
KEY="%ProGet.ApiKey%"
URL="http://proget.inc.elara.local/nuget/conan/package"
python3 - "$KEY" "$URL" <<'PY'
import sys, os, glob, http.client
from urllib.parse import urlparse
key = sys.argv[1]
u = urlparse(sys.argv[2])
B = "ProGetConanBoundary12345"
pkgs = sorted(glob.glob("all-nupkg/**/*.nupkg", recursive=True))
if not pkgs:
    print("no nupkg found")
    sys.exit(1)
rc = 0
for f in pkgs:
    fn = os.path.basename(f)
    cd = 'form-data; name="package"; filename="' + fn + '"'
    head = "--%s\r\nContent-Disposition: %s\r\n" % (B, cd)
    head += "Content-Type: application/octet-stream\r\n\r\n"
    head = head.encode()
    tail = ("\r\n--%s--\r\n" % B).encode()
    total = len(head) + os.path.getsize(f) + len(tail)
    try:
        c = http.client.HTTPConnection(u.hostname, u.port or 80)
        c.putrequest("PUT", u.path, skip_accept_encoding=True)
        c.putheader("X-ApiKey", key)
        c.putheader("X-NuGet-ApiKey", key)
        c.putheader("Content-Type", "multipart/form-data; boundary=" + B)
        c.putheader("Content-Length", str(total))
        c.endheaders()
        c.send(head)
        fh = open(f, "rb")
        while True:
            b = fh.read(1048576)
            if not b:
                break
            c.send(b)
        fh.close()
        c.send(tail)
        r = c.getresponse()
        code = r.status
        r.read()
        c.close()
    except Exception as e:
        print("ERR " + fn + " " + str(e))
        rc = 1
        continue
    tag = "OK"
    if code == 409:
        tag = "SKIP"
    elif code not in (200, 201):
        tag = "FAIL"
        rc = 1
    print("[" + str(code) + " " + tag + "] " + fn)
sys.exit(rc)
PY
