PLUG=$(find ~/.conan2 -type f -name grpc_cpp_plugin 2>/dev/null | head -1)
  echo "plugin: $PLUG"
  file "$PLUG"                                  # ждём ELF x86-64
  ldd "$PLUG" 2>/dev/null | grep -iE 'grpc|protobuf|not found' \
      || echo "статически слинкован — ок"

  # положить в lib/native/<suffix>/ нашего grpc-пакета (там же, где легаси держал плагин):
  G=~/grpc.lin.gcc84.shared.x86_64.1.60.1.1/lib/native/lin-gcc84-shared-x86_64
  cp "$PLUG" "$G"/grpc_cpp_plugin && chmod +x "$G"/grpc_cpp_plugin

  # перезапустить конфигурацию el_conf:
  cd ~/el_conf/.build/lin.gcc.shared.x64
  rm -f CMakeCache.txt && bash 1.sh
