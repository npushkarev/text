## Примечания

**Обновления бесплатны.** Новые открытые модели выходят раз в несколько месяцев — скачиваем и ставим на то же железо. Диск 4 ТБ рассчитан на хранение нескольких моделей одновременно.

**Отдельный человек не нужен.** Установка и сопровождение — силами команды инфраструктуры, на базе уже используемых у нас инструментов (Zabbix, nginx, домен).

**Данные не покидают сервер.** Ни один компонент не обращается к внешним сервисам в процессе работы. Внешний доступ нужен только на этапе установки и при скачивании новой модели.

**Что не ставим:** Kubernetes (избыточен для одного сервера), любые платные и облачные компоненты.

**Сроки:** установка и запуск — 1 неделя после монтажа, как заложено в плане внедрения.

#!/usr/bin/env python3
"""Прописывает NuGet.Config для сборки Elara-проектов на legacy CMake-фреймворке.

Фреймворк читает NuGet.Config сам, через cmake/NuGetProto.cmake. Без этого файла
configure падает на `NuGet config file "..." doesn't exists.`

Запуск:
    python setup_nuget_config.py              прописать файл и проверить фиды
    python setup_nuget_config.py --check      только проверить, ничего не менять
    python setup_nuget_config.py --insecure   не проверять TLS-сертификат ProGet
    python setup_nuget_config.py --print      показать путь к файлу и выйти

Работает и на Windows, и на Linux: путь к NuGet.Config выбирается по платформе.
Существующий файл не затирается молча, рядом кладётся копия с меткой времени.
"""

import argparse
import os
import re
import shutil
import ssl
import sys
import time
import urllib.error
import urllib.request

PROGET = "https://proget.inc.elara.local"

# Порядок важен: резолвер перебирает фиды сверху вниз.
FEEDS = [
    ("Default", PROGET + "/nuget/Default/"),
    ("Prerelease", PROGET + "/nuget/Prerelease/"),
    ("Legacy", PROGET + "/nuget/Legacy/"),
    ("External", PROGET + "/nuget/External/"),
    ("Conan", PROGET + "/nuget/Conan/"),
]

# Пакеты, которые ищем в фиде Conan для проверки. Слот win.v142.shared.x64
# соответствует BUILD_SHARED_LIBS=ON (дефолт легаси-проектов) и тулсету v142.
PROBE_PACKAGES = [
    "rapidjson.win.v142.shared.x64",
    "rapidjson.win.v142.static.x64",
]

TIMEOUT = 15


def config_path():
    if os.name == "nt":
        base = os.environ.get("APPDATA")
        if not base:
            sys.exit("Не задана переменная APPDATA, не могу определить путь к NuGet.Config")
        return os.path.join(base, "NuGet", "NuGet.Config")
    return os.path.join(os.path.expanduser("~"), ".config", "NuGet", "NuGet.Config")


def render():
    lines = [
        '<?xml version="1.0" encoding="utf-8"?>',
        "<configuration>",
        "  <packageSources>",
        "    <clear />",
    ]
    for key, url in FEEDS:
        lines.append('    <add key="%s" value="%s" />' % (key, url))
    lines += ["  </packageSources>", "</configuration>", ""]
    return "\n".join(lines)


def write_config(path):
    content = render()

    if os.path.exists(path):
        with open(path, "r", encoding="utf-8-sig") as handle:
            if handle.read() == content:
                print("[=] Файл уже в нужном виде: %s" % path)
                return False
        backup = "%s.%s.bak" % (path, time.strftime("%Y%m%d-%H%M%S"))
        shutil.copy2(path, backup)
        print("[i] Старый файл сохранён: %s" % backup)

    os.makedirs(os.path.dirname(path), exist_ok=True)
    # utf-8 без BOM: некоторые парсеры NuGet на BOM спотыкаются.
    with open(path, "w", encoding="utf-8", newline="\n") as handle:
        handle.write(content)
    print("[+] Записан %s" % path)
    return True


def opener(insecure):
    if not insecure:
        return urllib.request.build_opener()
    context = ssl.create_default_context()
    context.check_hostname = False
    context.verify_mode = ssl.CERT_NONE
    return urllib.request.build_opener(urllib.request.HTTPSHandler(context=context))


def fetch(client, url):
    request = urllib.request.Request(url, headers={"Accept": "application/xml"})
    with client.open(request, timeout=TIMEOUT) as response:
        return response.getcode(), response.read().decode("utf-8", "replace")


def check_feeds(client):
    ok = True
    for key, url in FEEDS:
        try:
            code, _ = fetch(client, url)
            print("[+] %-11s %s -> %s" % (key, url, code))
        except urllib.error.HTTPError as error:
            # 404 на корне фида бывает нормой, важно что сервер ответил.
            print("[~] %-11s %s -> HTTP %s" % (key, url, error.code))
        except ssl.SSLError as error:
            ok = False
            print("[!] %-11s %s -> ошибка TLS: %s" % (key, url, error))
            print("    Сертификат ProGet не в доверенных. Добавьте его в хранилище")
            print("    либо перезапустите с --insecure, чтобы отличить эту проблему от других.")
        except Exception as error:
            ok = False
            print("[!] %-11s %s -> %s" % (key, url, error))
    return ok


def check_packages(client):
    feed = dict(FEEDS)["Conan"]
    for package in PROBE_PACKAGES:
        url = "%sFindPackagesById()?id='%s'" % (feed, package)
        try:
            _, body = fetch(client, url)
        except Exception as error:
            print("[!] %-34s запрос не прошёл: %s" % (package, error))
            continue
        versions = re.findall(r"<d:Version>([^<]+)</d:Version>", body)
        if versions:
            print("[+] %-34s %s" % (package, ", ".join(sorted(set(versions)))))
        else:
            print("[!] %-34s в фиде Conan не найден" % package)


def check_toolchain():
    if shutil.which("nuget"):
        print("[+] nuget найден в PATH")
    else:
        print("[=] nuget в PATH нет. Фреймворку он и не нужен, фиды читает NuGetProto.cmake")
    if shutil.which("cmake"):
        print("[+] cmake найден в PATH")
    else:
        print("[!] cmake в PATH нет, configure запустить не получится")


def main():
    parser = argparse.ArgumentParser(description="Прописать NuGet.Config для legacy CMake-фреймворка")
    parser.add_argument("--check", action="store_true", help="только проверить, файл не трогать")
    parser.add_argument("--insecure", action="store_true", help="не проверять TLS-сертификат ProGet")
    parser.add_argument("--print", dest="show", action="store_true", help="показать путь и выйти")
    args = parser.parse_args()

    path = config_path()

    if args.show:
        print(path)
        return 0

    print("NuGet.Config: %s" % path)

    if args.check:
        if os.path.exists(path):
            print("[+] файл существует")
        else:
            print("[!] файла нет, запустите без --check")
    else:
        write_config(path)

    print()
    check_toolchain()

    print()
    client = opener(args.insecure)
    if check_feeds(client):
        print()
        check_packages(client)

    print()
    print("Дальше:")
    print("  cmake -S . -B build -DQT5_ROOT_DIR=\"C:/Qt/5.15.2/msvc2019_64\"")
    return 0


if __name__ == "__main__":
    sys.exit(main())
