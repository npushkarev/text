#!/bin/bash                                                                                                                         
  # Использование: source /tmp/setup_protoc_env.sh                                                                                    
                                                                                                                                      
  # 1. Собрать LD_LIBRARY_PATH из всех conan package folder'ов
  LD_PATHS=""                                                                                                                         
  for d in /home/user/.conan2/p/b/*/p/lib; do                                                                                         
      [ -d "$d" ] && LD_PATHS="$d:$LD_PATHS"
  done                                                                                                                                
                                                                                                                                    
  if [ -z "$LD_PATHS" ]; then                                                                                                         
      echo "ERROR: conan package folders not found in /home/user/.conan2/p/b/*/p/lib"                                               
      return 1 2>/dev/null || exit 1                                                                                                  
  fi
                                                                                                                                      
  export LD_LIBRARY_PATH="${LD_PATHS}${LD_LIBRARY_PATH}"                                                                              
   
  echo "=== LD_LIBRARY_PATH dirs (first 10): ==="                                                                                     
  echo "$LD_LIBRARY_PATH" | tr ':' '\n' | head -10                                                                                  
  echo                                                                                                                                
   
  # 2. Проверить grpc_cpp_plugin                                                                                                      
  echo "=== grpc_cpp_plugin ==="                                                                                                    
  if [ -x ~/bin/grpc_cpp_plugin ]; then                                                                                               
      missing=$(ldd ~/bin/grpc_cpp_plugin 2>&1 | grep "not found")
      if [ -z "$missing" ]; then                                                                                                      
          echo "OK: все зависимости найдены"                                                                                        
      else                                                                                                                            
          echo "MISSING:"                                                                                                           
          echo "$missing"                                                                                                             
      fi
  else                                                                                                                                
      echo "WARN: ~/bin/grpc_cpp_plugin не найден"                                                                                  
  fi
  echo                                                                                                                                
  
  # 3. Проверить protoc                                                                                                               
  echo "=== protoc ==="                                                                                                             
  if [ -x ~/bin/protoc ]; then
      missing_protoc=$(ldd ~/bin/protoc 2>&1 | grep "not found")
      if [ -z "$missing_protoc" ]; then                                                                                               
          echo "OK: все зависимости найдены"
          ~/bin/protoc --version                                                                                                      
      else                                                                                                                            
          echo "MISSING:"
          echo "$missing_protoc"                                                                                                      
      fi                                                                                                                            
  else
      echo "WARN: ~/bin/protoc не найден"
  fi                                                                                                                                  
  echo
                                                                                                                                      
  # 4. PATH                                                                                                                         
  export PATH=~/bin:$PATH
  echo "=== PATH: ==="
  which protoc                                                                                                                        
  which grpc_cpp_plugin
