(async () => {
    const API = "/app/rest/2018.1";
    const sourcePath = ["SURA2", "COMPONENTS", "CMAKE", "EL_CONF"];

    const repos = [
      "as_system_data",
      "el_conf_chart_data_manager",
      "el_conf_config_builder",
      "el_conf_device_config_manager",
      "el_conf_core_sdk",
      "el_conf_gui_sdk",
      "el_conf_themes",
      "el_conf_utils",
      "el_conf_elecont_protocol_client",
      "el_conf_plugin_sdk",
      "el_conf_sura_connector_client",
      "el_conf_chart_ui_plugin",
      "as_app"
    ];

    const xmlEscape = value => String(value).replace(
      /[&<>"']/g,
      ch => ({
        "&": "&amp;",
        "<": "&lt;",
        ">": "&gt;",
        '"': "&quot;",
        "'": "&apos;"
      })[ch]
    );

    async function request(
      method,
      path,
      body = null,
      contentType = null,
      accept = "application/json"
    ) {
      const headers = { Accept: accept };
      if (contentType) headers["Content-Type"] = contentType;

      const response = await fetch(API + path, {
        method,
        headers,
        body,
        credentials: "same-origin"
      });

      const text = await response.text();

      if (!response.ok) {
        throw new Error(
          `${method} ${path}: HTTP ${response.status}\n${text}`
        );
      }

      return text;
    }

    async function getJson(path) {
      return JSON.parse(await request("GET", path));
    }

    const query = new URLSearchParams({
      locator: "count:10000",
      fields: "project(id,name,parentProjectId,description)"
    });

    const allProjects =
      (await getJson("/projects?" + query)).project || [];

    const byId = new Map(allProjects.map(project => [project.id, project]));

    function fullPath(project) {
      const result = [];
      const visited = new Set();

      while (project && project.id !== "_Root") {
        if (visited.has(project.id)) {
          throw new Error(`Цикл в дереве проектов: ${project.id}`);
        }

        visited.add(project.id);
        result.unshift(project.name);
        project = byId.get(project.parentProjectId);
      }

      return result;
    }

    const wanted = sourcePath.join("/").toLowerCase();

    const matches = allProjects.filter(project =>
      fullPath(project).join("/").toLowerCase() === wanted
    );

    if (matches.length !== 1) {
      throw new Error(
        `Проект ${sourcePath.join(" / ")} найден ${matches.length} раз`
      );
    }

    const source = matches[0];
    const parentId = source.parentProjectId;

    const children = allProjects.filter(
      project => project.parentProjectId === parentId
    );

    const childrenByName = new Map(
      children.map(project => [project.name.toLowerCase(), project])
    );

    const childrenById = new Map(
      children.map(project => [project.id, project])
    );

    function idPart(name) {
      return name
        .split(/[^A-Za-z0-9]+/)
        .filter(Boolean)
        .map(word => word[0].toUpperCase() + word.slice(1))
        .join("");
    }

    const plan = repos.map(repo => {
      const id = `${parentId}_${idPart(repo)}`;
      const sameName = childrenByName.get(repo.toLowerCase());
      const sameId = childrenById.get(id);

      let state = "CREATE";

      if (
        (sameName && sameName.id !== id) ||
        (sameId && sameId.name.toLowerCase() !== repo.toLowerCase())
      ) {
        state = "CONFLICT";
      } else if (sameName || sameId) {
        state = "EXISTS";
      }

      return { repo, id, state };
    });

    console.table(plan);

    if (plan.some(item => item.state === "CONFLICT")) {
      throw new Error("Есть CONFLICT. Ничего не создано.");
    }

    const createCount = plan.filter(
      item => item.state === "CREATE"
    ).length;

    if (!window.confirm(
      `Создать ${createCount} проектов и настроить все 13?\n\n` +
      "Источник: SURA2 / COMPONENTS / CMAKE / EL_CONF"
    )) {
      console.log("Отменено. TeamCity не изменён.");
      return;
    }

    async function setParameters(projectId, repo) {
      const locator = encodeURIComponent(`id:${projectId}`);
      const parameterQuery = new URLSearchParams({
        fields: "property(name)"
      });

      const parameters = await getJson(
        `/projects/${locator}/parameters?${parameterQuery}`
      );

      const existing = new Set(
        (parameters.property || []).map(item => item.name)
      );

      const desired = {
        repoProject: "su2",
        repoName: repo,
        projectName: repo,
        "iac.managedBy": "su2-repos-setup/teamcity-f12"
      };

      for (const [name, value] of Object.entries(desired)) {
        if (existing.has(name)) {
          await request(
            "PUT",
            `/projects/${locator}/parameters/${encodeURIComponent(name)}`,
            value,
            "text/plain; charset=utf-8",
            "text/plain"
          );
        } else {
          const property =
            `<?xml version="1.0" encoding="UTF-8"?>` +
            `<property name="${xmlEscape(name)}" ` +
            `value="${xmlEscape(value)}"/>`;

          await request(
            "POST",
            `/projects/${locator}/parameters`,
            property,
            "application/xml"
          );
        }
      }
    }

    for (const item of plan) {
      if (item.state === "CREATE") {
        console.log(`Создаю ${item.repo}...`);

        const payload =
          `<?xml version="1.0" encoding="UTF-8"?>` +
          `<newProjectDescription ` +
          `copyAllAssociatedSettings="true" ` +
          `id="${xmlEscape(item.id)}" ` +
          `name="${xmlEscape(item.repo)}" ` +
          `description="Managed by su2-repos-setup/teamcity-f12">` +
          `<sourceProject locator="id:${xmlEscape(source.id)}"/>` +
          `<parentProject locator="id:${xmlEscape(parentId)}"/>` +
          `</newProjectDescription>`;

        await request(
          "POST",
          "/projects",
          payload,
          "application/xml"
        );
      }

      await setParameters(item.id, item.repo);
      console.log(`OK: ${item.repo}`);
    }

    console.log("ГОТОВО: настроены все 13 проектов.");
  })().catch(error => console.error("ОШИБКА:", error));
