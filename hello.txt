for nupkg in *.nupkg; do                                                                                                                                             
      base="${nupkg%.nupkg}"                                                                                                                                         
      target="$HOME/$base"
      rm -rf "$target"
      mkdir -p "$target"                                                                                                                                               
      cp "$nupkg" "$target/$nupkg"          # .nupkg сам как файл
      unzip -o "$nupkg" -d "$target" > /dev/null   # И распаковать рядом                                                                                               
      echo "✓ $nupkg → $target/ (файл + распаковка)"                                                                                                                 
  done
