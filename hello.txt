#!/bin/bash
#
# deps_report.sh — отчёт по дереву зависимостей legacy NuGet пакетов.
#
# Назначение:
#   1. Показать прямые/транзитивные зависимости каждого pkg в NuGet-кэше.
#   2. Показать обратные зависимости (кто использует pkg-кандидат на миграцию).
#   3. Найти outdated transitive deps (protobuf 4.x, openssl 1.1.x, zlib 1.3.0).
#
# Использование:
#   ./deps_report.sh                          — полный отчёт (forward + reverse)
#   ./deps_report.sh flat                     — карта pkg → его _dependencies
#   ./deps_report.sh tree <pkg> <ver>         — forward-дерево от pkg вниз
#   ./deps_report.sh reverse <pkg> <ver>      — reverse-дерево: кто потребляет pkg
#   ./deps_report.sh migration                — reverse-деревья для всех outdated
#   ./deps_report.sh outdated                 — список outdated entries в кэше
#   ./deps_report.sh el_conf                  — forward-дерево от el_conf вниз
#   ./deps_report.sh el_conf-reverse          — reverse-tree для каждой top-dep el_conf
#   ./deps_report.sh el_conf-graph            — forward+reverse для каждой top-dep el_conf
#
# Переменные окружения:
#   NUGET_DIR    — путь к NuGet-кэшу (default: $HOME/.nuget/packages)
#   EL_CONF_DIR  — путь к корню el_conf  (default: $HOME/el_conf)
#
# Примеры:
#   ./deps_report.sh > /tmp/report.txt
#   ./deps_report.sh tree mosquitto 2.0.19
#   ./deps_report.sh reverse openssl 1.1.11        # кто тянет старый openssl
#   ./deps_report.sh reverse protobuf 4.25.2       # кто тянет старый protobuf
#   ./deps_report.sh migration > /tmp/owners.txt
#
set -euo pipefail

NUGET_DIR="${NUGET_DIR:-$HOME/.nuget/packages}"
EL_CONF_DIR="${EL_CONF_DIR:-$HOME/el_conf}"

# Пакеты-кандидаты на миграцию (обновить при необходимости).
MIGRATION_CANDIDATES=(
    "openssl:1.1.11"
    "protobuf:4.25.2"
    "zlib:1.3.0"
    "mosquitto:2.0.19"
    "gsd_parser:0.15.0"
    "grpc:1.60.1"
)

# ---------- helpers -----------------------------------------------------------

# Найти CMakeLists.var для конкретной пары pkg+version в NuGet-кэше.
find_var_file() {
    local pkg="$1"
    local ver="$2"
    find "$NUGET_DIR" -path "*${pkg}.lin.*/${ver}/CMakeLists.var" 2>/dev/null | head -1
}

# Извлечь блок _dependencies из CMakeLists.var, по одной "name:version" на строку.
extract_deps() {
    local var_file="$1"
    [ -f "$var_file" ] || return 0
    awk '
        /set\(\$\{project_name\}_dependencies/ { in_block=1; next }
        in_block && /^[[:space:]]*\)/ { in_block=0; exit }
        in_block {
            gsub(/^[[:space:]]+|[[:space:]]+$/, "")
            if (length($0) > 0) print
        }
    ' "$var_file"
}

# Превратить путь CMakeLists.var в "name:version" consumer'a.
# В NuGet-кэше: <root>/<name>.lin.gcc84.shared.x86_64/<ver>/CMakeLists.var
# el_conf репо:  $EL_CONF_DIR/CMakeLists.var → "el_conf:source"
consumer_of() {
    local var_file="$1"
    if [[ "$var_file" == "$EL_CONF_DIR/"* ]]; then
        echo "el_conf:source"
        return
    fi
    local rel="${var_file#$NUGET_DIR/}"
    local pkg_dir="${rel%%/*}"
    local pkg_name="${pkg_dir%%.lin.*}"
    local rest="${rel#$pkg_dir/}"
    local pkg_ver="${rest%%/*}"
    echo "${pkg_name}:${pkg_ver}"
}

# Все CMakeLists.var в кэше + (если есть) el_conf/CMakeLists.var.
all_var_files() {
    find "$NUGET_DIR" -name CMakeLists.var 2>/dev/null
    [ -f "$EL_CONF_DIR/CMakeLists.var" ] && echo "$EL_CONF_DIR/CMakeLists.var"
}

# Построить ассоциативную мапу: dep "name:ver" → пробел-разделённый список consumers.
declare -A CONSUMERS_MAP
CONSUMERS_BUILT=0

build_consumers_map() {
    [ "$CONSUMERS_BUILT" = "1" ] && return
    local var_file consumer dep
    while IFS= read -r var_file; do
        [ -f "$var_file" ] || continue
        consumer="$(consumer_of "$var_file")"
        while IFS= read -r dep; do
            [ -z "$dep" ] && continue
            CONSUMERS_MAP["$dep"]="${CONSUMERS_MAP[$dep]:-} $consumer"
        done < <(extract_deps "$var_file")
    done < <(all_var_files)
    CONSUMERS_BUILT=1
}

# ---------- forward (что pkg тянет вниз) --------------------------------------

# Рекурсивное дерево вниз от заданного pkg+version.
deps_tree() {
    local pkg="$1"
    local ver="$2"
    local prefix="${3:-}"
    local seen="${4:-}"

    echo "${prefix}${pkg}:${ver}"

    case "$seen" in
        *"|${pkg}:${ver}|"*)
            echo "${prefix}    └── (already shown — cycle or duplicate)"
            return
            ;;
    esac
    seen="${seen}|${pkg}:${ver}|"

    local var_file
    var_file=$(find_var_file "$pkg" "$ver")
    if [ -z "$var_file" ] || [ ! -f "$var_file" ]; then
        echo "${prefix}    └── (NOT IN CACHE)"
        return
    fi

    local deps
    deps=$(extract_deps "$var_file")
    [ -z "$deps" ] && return

    local dep
    while IFS= read -r dep; do
        [ -z "$dep" ] && continue
        local name="${dep%%:*}"
        local v="${dep##*:}"
        deps_tree "$name" "$v" "${prefix}    " "$seen"
    done <<< "$deps"
}

# ---------- reverse (кто использует pkg) --------------------------------------

# Рекурсивное reverse-дерево: показывает consumers'ов pkg+ver, потом их consumers'ов.
deps_reverse_tree() {
    local target="$1:$2"
    local prefix="${3:-}"
    local seen="${4:-}"

    echo "${prefix}${target}"

    case "$seen" in
        *"|$target|"*)
            echo "${prefix}    ↑ (already shown)"
            return
            ;;
    esac
    seen="${seen}|$target|"

    build_consumers_map

    local list="${CONSUMERS_MAP[$target]:-}"
    [ -z "$list" ] && { echo "${prefix}    (no known consumers)"; return; }

    # Уникальные consumers (один pkg может несколько раз попасть).
    local consumer
    for consumer in $(echo "$list" | tr ' ' '\n' | sort -u | grep -v '^$'); do
        deps_reverse_tree "${consumer%:*}" "${consumer##*:}" "${prefix}    ↑ " "$seen"
    done
}

# ---------- модули отчёта ------------------------------------------------------

deps_flat() {
    echo "========================================="
    echo "Flat dependency map of $NUGET_DIR"
    echo "========================================="
    echo
    local var_file
    for var_file in $(find "$NUGET_DIR" -name CMakeLists.var 2>/dev/null | sort); do
        local consumer
        consumer="$(consumer_of "$var_file")"
        echo "=== $consumer ==="
        local deps
        deps=$(extract_deps "$var_file")
        if [ -z "$deps" ]; then
            echo "    (no dependencies)"
        else
            echo "$deps" | sed 's/^/    /'
        fi
        echo
    done
}

deps_outdated() {
    echo "========================================="
    echo "Outdated transitive deps blocking migration"
    echo "========================================="
    echo "(searching in $NUGET_DIR)"
    echo
    find "$NUGET_DIR" -name CMakeLists.var 2>/dev/null \
        | xargs grep -HE "protobuf:[0-9]|openssl:[0-9]|zlib:[0-9]" 2>/dev/null \
        | grep -E ":4\.|:1\.1\.|zlib:1\.3\.0" \
        | sed "s|${NUGET_DIR}/||g"
}

deps_el_conf_top() {
    local var_file="${EL_CONF_DIR}/CMakeLists.var"
    echo "========================================="
    echo "el_conf top-level dependencies"
    echo "========================================="
    echo "(from $var_file)"
    echo
    if [ ! -f "$var_file" ]; then
        echo "    NOT FOUND: $var_file"
        return
    fi
    extract_deps "$var_file" | sed 's/^/    /'
}

# ---------- el_conf-specific отчёты -------------------------------------------

# Список top-level deps el_conf'a в виде "name:version" по одному на строку.
el_conf_top_deps_list() {
    local var_file="${EL_CONF_DIR}/CMakeLists.var"
    [ -f "$var_file" ] || return 0
    extract_deps "$var_file"
}

# Forward-дерево от el_conf вниз: top-level deps + рекурсивный обход каждой.
el_conf_forward_tree() {
    echo "========================================="
    echo "el_conf full forward dependency tree"
    echo "========================================="
    echo "el_conf:source"
    local dep
    while IFS= read -r dep; do
        [ -z "$dep" ] && continue
        local name="${dep%%:*}"
        local ver="${dep##*:}"
        deps_tree "$name" "$ver" "    "
    done < <(el_conf_top_deps_list)
}

# Reverse-tree для каждой top-dep el_conf'a (кто ещё её использует кроме el_conf).
el_conf_reverse_for_each() {
    echo "========================================="
    echo "Reverse-tree for each el_conf top-level dep"
    echo "========================================="
    echo "Для каждой прямой зависимости el_conf — кто ещё её потребляет в кэше."
    echo
    local dep
    while IFS= read -r dep; do
        [ -z "$dep" ] && continue
        echo "### Consumers of $dep"
        deps_reverse_tree "${dep%%:*}" "${dep##*:}"
        echo
    done < <(el_conf_top_deps_list)
}

# Богатый отчёт: forward (что тянет) + reverse (кто ещё тянет) для каждой top-dep.
el_conf_graph_report() {
    echo "========================================="
    echo "el_conf dependency graph — forward + reverse for each top-level dep"
    echo "========================================="
    echo
    local dep
    while IFS= read -r dep; do
        [ -z "$dep" ] && continue
        local name="${dep%%:*}"
        local ver="${dep##*:}"
        echo "######## $dep ########"
        echo
        echo "--- forward (what '$dep' pulls in) ---"
        deps_tree "$name" "$ver"
        echo
        echo "--- reverse (who else consumes '$dep') ---"
        deps_reverse_tree "$name" "$ver"
        echo
    done < <(el_conf_top_deps_list)
}

# Reverse-деревья для всех pkg-кандидатов на миграцию.
migration_consumers_report() {
    echo "========================================="
    echo "Migration consumers report"
    echo "========================================="
    echo "Для каждого устаревшего пакета — дерево потребителей вверх."
    echo "↑ означает 'используется кем-то выше' (consumer)."
    echo
    local entry
    for entry in "${MIGRATION_CANDIDATES[@]}"; do
        echo "### Consumers of $entry"
        deps_reverse_tree "${entry%:*}" "${entry##*:}"
        echo
    done
}

full_report() {
    echo "==============================================="
    echo "Legacy NuGet migration dependency report"
    echo "Generated: $(date)"
    echo "Host:      $(hostname)"
    echo "NuGet dir: $NUGET_DIR"
    echo "==============================================="
    echo

    deps_el_conf_top
    echo

    deps_flat
    echo

    echo "========================================="
    echo "Forward trees for key roots"
    echo "========================================="
    echo
    local entry
    for entry in \
            "grpc_sdk:1.2.0" \
            "cs_system_data:1.132.0" \
            "cs_service_sdk:3.16.0" \
            "mosquitto:2.0.19" \
            "sura_proto:7.7.0" \
            "gsd_parser:0.15.0" \
            "ipc_factories_grpc:2.12.0" \
            "service_sdk:3.9.0"; do
        local name="${entry%%:*}"
        local ver="${entry##*:}"
        echo "### Tree: ${name}:${ver}"
        deps_tree "$name" "$ver"
        echo
    done

    migration_consumers_report
    echo

    deps_outdated
}

main() {
    local cmd="${1:-full}"
    case "$cmd" in
        full)
            full_report
            ;;
        flat)
            deps_flat
            ;;
        tree)
            local pkg="${2:-}"
            local ver="${3:-}"
            if [ -z "$pkg" ] || [ -z "$ver" ]; then
                echo "Usage: $0 tree <pkg_name> <version>" >&2
                exit 1
            fi
            deps_tree "$pkg" "$ver"
            ;;
        reverse)
            local pkg="${2:-}"
            local ver="${3:-}"
            if [ -z "$pkg" ] || [ -z "$ver" ]; then
                echo "Usage: $0 reverse <pkg_name> <version>" >&2
                exit 1
            fi
            deps_reverse_tree "$pkg" "$ver"
            ;;
        migration)
            migration_consumers_report
            ;;
        outdated)
            deps_outdated
            ;;
        el_conf)
            el_conf_forward_tree
            ;;
        el_conf-reverse)
            el_conf_reverse_for_each
            ;;
        el_conf-graph)
            el_conf_graph_report
            ;;
        *)
            echo "Usage: $0 [full|flat|tree <pkg> <ver>|reverse <pkg> <ver>|migration|outdated|el_conf|el_conf-reverse|el_conf-graph]" >&2
            exit 1
            ;;
    esac
}

main "$@"
